  575                 ItemStack itemstack = this.getInventory().getItem(i);
  576                 if (itemstack.getItem().isComplex()) {
  577                     Packet<?> packet = ((ComplexItem)itemstack.getItem()).getUpdatePacket(itemstack, this.level(), this);
  578                     if (packet != null) {
  579                         this.connection.send(packet);
  580                     }
  581                 }
  582             }
  583 
  584             if (this.getHealth() != this.lastSentHealth
  585                 || this.lastSentFood != this.foodData.getFoodLevel()
  586                 || this.foodData.getSaturationLevel() == 0.0F != this.lastFoodSaturationZero) {
  587                 this.connection.send(new ClientboundSetHealthPacket(this.getHealth(), this.foodData.getFoodLevel(), this.foodData.getSaturationLevel()));
